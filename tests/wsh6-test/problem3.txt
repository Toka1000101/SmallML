let fun fib(n) =
	if n = 0 then 0
	else if n = 1 then 1
	else
		let val a = 0 in
			let val b = 1 in
				let val i = 1 in
					let val next = 0 in
					{
						while i < n do {
							next := a + b;
							a := b;
							b := next;
							i := i + 1
						};
						b
					}
				end
			end
		end
	end
in
	fib(5)
end;
